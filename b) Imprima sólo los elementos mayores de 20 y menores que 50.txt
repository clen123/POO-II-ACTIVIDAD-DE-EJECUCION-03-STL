#include <array>
#include <iostream>
#include <algorithm>
using namespace std;

int main()
{
    array<int, 20> arr = {21, 23, 25, 27, 29, 11, 13, 15, 17, 19, 31, 33, 35, 37,                 39,1, 3, 5, 7, 9};
 
    cout << "Elementos mayores de 20 y menores que 50: " << endl;
    for (auto x : arr)
    {
        if (x > 20 && x < 50)
        {
            cout << x << " ";
        }
    }
    return 0;
}
